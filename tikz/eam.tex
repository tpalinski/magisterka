\usetikzlibrary{positioning,arrows.meta,fit}

\begin{tikzpicture}[
    scale=0.65,
    transform shape,
    font=\small,
    block/.style={
        draw,
        rounded corners=2pt,
        thick,
        minimum width=2.4cm,
        minimum height=0.85cm,
        align=center
    },
    enc/.style={block, fill=blue!20},
    trans/.style={block, fill=purple!20},
    attn/.style={block, fill=orange!25},
    proj/.style={block, fill=green!20},
    arrow/.style={->, thick}
]

\newcommand{\transformerstack}[3]{%
    % Invisible node defining the stack's bounding box
    \node[
        minimum width=2.7cm,
        minimum height=1.6cm,
        inner sep=0pt
    ] (#3) at (#1,#2) {};

    % Back layers
    \foreach \i in {3,2,1} {
        \draw[
            fill=purple!12,
            draw=purple!60!black,
            thick
        ]
        (#1-1.2+\i*0.10,#2-0.375+\i*0.10)
        rectangle
        (#1+1.2+\i*0.10,#2+0.375+\i*0.10);
    }

    % Front layer
    \draw[
        fill=purple!25,
        draw=purple!60!black,
        thick,
        rounded corners=1pt
    ]
    (#1-1.2,#2-0.375)
    rectangle
    (#1+1.2,#2+0.375);

    % Label
    \node[
        align=center,
        text width=2.1cm
    ] at (#1,#2)
    {Transformer\\Encoder};

    % × N
    \node[
        font=\small\bfseries
    ] at (#1,#2-0.72)
    {$\times N$};
}

% ============================================================

\node[block] (text) {Text Input};

\node[enc, right=1.2cm of text] (txtenc)
    {Pretrained\\Text Encoder};
